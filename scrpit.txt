-- =================================
-- ELIMINAR TABLAS SI YA EXISTEN
-- =================================
-- El orden es importante por las FK.

DROP TABLE IF EXISTS documento;
DROP TABLE IF EXISTS pago;
DROP TABLE IF EXISTS matricula;
DROP TABLE IF EXISTS grado_seccion;
DROP TABLE IF EXISTS estudiante;
DROP TABLE IF EXISTS apoderado;


-- =================================
-- TABLA: APODERADO
-- =================================

CREATE TABLE apoderado (
    id SERIAL PRIMARY KEY,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    dni VARCHAR(8) NOT NULL UNIQUE,
    telefono VARCHAR(15),
    email VARCHAR(100)
);


-- =================================
-- TABLA: ESTUDIANTE
-- =================================

CREATE TABLE estudiante (
    id SERIAL PRIMARY KEY,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    fecha_nac DATE NOT NULL,
    direccion VARCHAR(200),
    dni VARCHAR(8) NOT NULL UNIQUE,
    id_apoderado INTEGER,

    FOREIGN KEY (id_apoderado)
        REFERENCES apoderado(id)
);


-- =================================
-- TABLA: GRADO_SECCION
-- =================================

CREATE TABLE grado_seccion (
    id SERIAL PRIMARY KEY,
    grado VARCHAR(20) NOT NULL,
    seccion CHAR(1) NOT NULL
);


-- =================================
-- TABLA: MATRICULA
-- =================================

CREATE TABLE matricula (
    id SERIAL PRIMARY KEY,
    anio INTEGER NOT NULL,
    estado VARCHAR(20) NOT NULL,
    fecha DATE NOT NULL,
    id_estudiante INTEGER NOT NULL,
    id_grado_seccion INTEGER NOT NULL,

    FOREIGN KEY (id_estudiante)
        REFERENCES estudiante(id),

    FOREIGN KEY (id_grado_seccion)
        REFERENCES grado_seccion(id)
);


-- =================================
-- TABLA: PAGO
-- =================================

CREATE TABLE pago (
    id SERIAL PRIMARY KEY,
    fecha DATE NOT NULL,
    monto NUMERIC(10,2) NOT NULL,
    comprobante VARCHAR(50),
    id_matricula INTEGER NOT NULL,

    FOREIGN KEY (id_matricula)
        REFERENCES matricula(id)
);


-- =================================
-- TABLA: DOCUMENTO
-- =================================

CREATE TABLE documento (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    ruta VARCHAR(255),
    id_estudiante INTEGER NOT NULL,

    FOREIGN KEY (id_estudiante)
        REFERENCES estudiante(id)
);
